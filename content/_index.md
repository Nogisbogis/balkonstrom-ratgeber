[[faq]]
  f = 'Was ist der Unterschied zwischen Kapazität und Dauerleistung?'
  a = 'Die Kapazität in Wattstunden bestimmt, wie lange etwas läuft. Die Dauerleistung in Watt bestimmt, was überhaupt läuft: Sie muss höher sein als die Leistung aller Geräte, die gleichzeitig angeschlossen sind. Mehr dazu in der [Camping-Kaufberatung](/posts/powerstation-camping-kaufberatung/).'
[[faq]]
  f = 'Wie viel Kapazität brauche ich?'
  a = 'Rechne für jedes Gerät Watt × Stunden und addiere alles. Plane einen Puffer ein, weil beim Umwandeln auf 230 V Energie verloren geht; der ADAC stellte 2025 fest, dass die Powerstations weniger lieferten als angegeben. Der [Laufzeitrechner](/laufzeitrechner/) nimmt dir das Rechnen ab.'
[[faq]]
  f = 'Kann ich mit einer Powerstation bei Stromausfall mein Haus versorgen?'
  a = 'Nicht über die Steckdose. Eine Powerstation versorgt die Geräte, die du direkt anschließt. Wer ins Hausnetz einspeisen will, muss laut BBK einen Elektrofachbetrieb beauftragen. Mehr im Artikel [Powerstation als Notstrom](/posts/powerstation-notstrom-blackout/).'
[[faq]]
  f = 'Wann lohnt sich ein Speicher fürs Balkonkraftwerk?'
  a = 'Am ehesten, wenn deine Module tagsüber regelmäßig deutlich mehr erzeugen, als du gleichzeitig verbrauchst. In meiner Beispielrechnung (nur Annahmen: 1,92 kWh, 200 volle Zyklen, 90 %, 35 ct/kWh) amortisiert sich der Speicher allein nach etwa sechs bis acht Jahren. Deinen Fall rechnest du mit dem [Amortisationsrechner](/amortisationsrechner/).'
[[faq]]
  f = 'Wo muss ich mein Balkonkraftwerk anmelden?'
  a = 'Nur im Marktstammdatenregister der Bundesnetzagentur, innerhalb eines Monats nach Inbetriebnahme. Die frühere Meldung beim Netzbetreiber entfällt; einen Speicher musst du laut Bundeswirtschaftsministerium aber zusätzlich beim Netzbetreiber anmelden. Alles zum Ablauf: [Balkonkraftwerk anmelden](/posts/balkonkraftwerk-anmelden-2026/).'
[[faq]]
  f = 'Darf ich eine Powerstation mit ins Flugzeug nehmen?'
  a = 'Keine der hier verglichenen. Lufthansa verbietet Akkus über 160 Wh (Ausnahmen: Mobilitätshilfen und medizinische Geräte), und die kleinste Station im Vergleich hat 245 Wh.'
